# 应用场景

## Web 服务器

多线程处理并发客户端连接，每个请求由一个线程处理。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <pthread.h>
#include <sys/socket.h>
#include <netinet/in.h>

#define PORT 8080
#define MAX_THREADS 100

void *handle_client(void *arg)
{
    int client_fd = *(int *)arg;
    free(arg);

    char buffer[1024];
    int n = read(client_fd, buffer, sizeof(buffer) - 1);
    if (n > 0) {
        buffer[n] = '\0';
        printf("Received: %s\n", buffer);

        // 发送响应
        const char *response = "HTTP/1.1 200 OK\r\n\r\nHello";
        write(client_fd, response, strlen(response));
    }

    close(client_fd);
    return NULL;
}

int main(void)
{
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    struct sockaddr_in addr = {
        .sin_family = AF_INET,
        .sin_port = htons(PORT),
        .sin_addr.s_addr = INADDR_ANY
    };

    bind(server_fd, (struct sockaddr *)&addr, sizeof(addr));
    listen(server_fd, 10);

    printf("Server listening on port %d\n", PORT);

    while (1) {
        struct sockaddr_in client_addr;
        socklen_t addr_len = sizeof(client_addr);
        int client_fd = accept(server_fd,
                               (struct sockaddr *)&client_addr,
                               &addr_len);

        int *fd_ptr = malloc(sizeof(int));
        *fd_ptr = client_fd;

        pthread_t tid;
        pthread_create(&tid, NULL, handle_client, fd_ptr);
        pthread_detach(tid);  // 分离线程，自动回收
    }

    close(server_fd);
    return 0;
}
```

> **注意**：生产环境应使用线程池限制并发数，避免资源耗尽。

## 生产者-消费者模型

多线程协作处理数据流，典型应用：任务队列、日志系统、消息处理。

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

#define QUEUE_SIZE 100

typedef struct {
    void *items[QUEUE_SIZE];
    int front, rear, count;
    pthread_mutex_t mutex;
    pthread_cond_t not_full;
    pthread_cond_t not_empty;
} TaskQueue;

void queue_init(TaskQueue *q)
{
    q->front = q->rear = q->count = 0;
    pthread_mutex_init(&q->mutex, NULL);
    pthread_cond_init(&q->not_full, NULL);
    pthread_cond_init(&q->not_empty, NULL);
}

void queue_push(TaskQueue *q, void *item)
{
    pthread_mutex_lock(&q->mutex);
    while (q->count == QUEUE_SIZE) {
        pthread_cond_wait(&q->not_full, &q->mutex);
    }
    q->items[q->rear] = item;
    q->rear = (q->rear + 1) % QUEUE_SIZE;
    q->count++;
    pthread_cond_signal(&q->not_empty);
    pthread_mutex_unlock(&q->mutex);
}

void *queue_pop(TaskQueue *q)
{
    pthread_mutex_lock(&q->mutex);
    while (q->count == 0) {
        pthread_cond_wait(&q->not_empty, &q->mutex);
    }
    void *item = q->items[q->front];
    q->front = (q->front + 1) % QUEUE_SIZE;
    q->count--;
    pthread_cond_signal(&q->not_full);
    pthread_mutex_unlock(&q->mutex);
    return item;
}
```

## 并行计算

将大任务拆分为子任务，利用多核 CPU 加速计算。

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

#define NUM_THREADS 4
#define ARRAY_SIZE 10000000

static double array[ARRAY_SIZE];
static double partial_sums[NUM_THREADS];

void *sum_chunk(void *arg)
{
    int id = *(int *)arg;
    int chunk = ARRAY_SIZE / NUM_THREADS;
    int start = id * chunk;
    int end = (id == NUM_THREADS - 1) ? ARRAY_SIZE : start + chunk;

    partial_sums[id] = 0;
    for (int i = start; i < end; i++) {
        partial_sums[id] += array[i];
    }

    return NULL;
}

int main(void)
{
    // 初始化数据
    for (int i = 0; i < ARRAY_SIZE; i++) {
        array[i] = i + 1;
    }

    pthread_t threads[NUM_THREADS];
    int ids[NUM_THREADS];

    for (int i = 0; i < NUM_THREADS; i++) {
        ids[i] = i;
        pthread_create(&threads[i], NULL, sum_chunk, &ids[i]);
    }

    double total = 0;
    for (int i = 0; i < NUM_THREADS; i++) {
        pthread_join(threads[i], NULL);
        total += partial_sums[i];
    }

    printf("Total sum: %.0f\n", total);
    return 0;
}
```

## 实时数据处理

传感器数据采集、音视频处理等需要及时响应的场景。

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <unistd.h>

#define BUFFER_SIZE 1024

static double sensor_buffer[BUFFER_SIZE];
static int buffer_index = 0;
static pthread_mutex_t buffer_mutex = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t data_ready = PTHREAD_COND_INITIALIZER;

// 模拟传感器读取线程
void *sensor_reader(void *arg)
{
    while (1) {
        // 模拟读取传感器数据
        double value = (double)rand() / RAND_MAX * 100;

        pthread_mutex_lock(&buffer_mutex);
        sensor_buffer[buffer_index++] = value;
        if (buffer_index >= BUFFER_SIZE) {
            buffer_index = 0;
            pthread_cond_signal(&data_ready);  // 通知处理线程
        }
        pthread_mutex_unlock(&buffer_mutex);

        usleep(1000);  // 1ms 采样间隔
    }
    return NULL;
}

// 数据处理线程
void *data_processor(void *arg)
{
    while (1) {
        pthread_mutex_lock(&buffer_mutex);
        pthread_cond_wait(&data_ready, &buffer_mutex);

        // 处理缓冲区数据
        double sum = 0;
        for (int i = 0; i < BUFFER_SIZE; i++) {
            sum += sensor_buffer[i];
        }
        double average = sum / BUFFER_SIZE;
        printf("Average: %.2f\n", average);

        pthread_mutex_unlock(&buffer_mutex);
    }
    return NULL;
}

int main(void)
{
    pthread_t reader, processor;

    pthread_create(&reader, NULL, sensor_reader, NULL);
    pthread_create(&processor, NULL, data_processor, NULL);

    pthread_join(reader, NULL);
    pthread_join(processor, NULL);

    return 0;
}
```

## 线程池

管理固定数量的工作线程，避免频繁创建销毁线程的开销。

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <unistd.h>

#define POOL_SIZE 4
#define QUEUE_SIZE 100

typedef struct {
    void (*function)(void *);
    void *argument;
} Task;

typedef struct {
    Task queue[QUEUE_SIZE];
    int front, rear, count;
    pthread_mutex_t mutex;
    pthread_cond_t not_empty;
    pthread_cond_t not_full;
    int shutdown;
} ThreadPool;

static ThreadPool pool;

void *worker_thread(void *arg)
{
    while (1) {
        pthread_mutex_lock(&pool.mutex);

        while (pool.count == 0 && !pool.shutdown) {
            pthread_cond_wait(&pool.not_empty, &pool.mutex);
        }

        if (pool.shutdown) {
            pthread_mutex_unlock(&pool.mutex);
            break;
        }

        Task task = pool.queue[pool.front];
        pool.front = (pool.front + 1) % QUEUE_SIZE;
        pool.count--;
        pthread_cond_signal(&pool.not_full);

        pthread_mutex_unlock(&pool.mutex);

        // 执行任务
        task.function(task.argument);
    }
    return NULL;
}

void pool_init(void)
{
    pool.front = pool.rear = pool.count = 0;
    pool.shutdown = 0;
    pthread_mutex_init(&pool.mutex, NULL);
    pthread_cond_init(&pool.not_empty, NULL);
    pthread_cond_init(&pool.not_full, NULL);

    pthread_t threads[POOL_SIZE];
    for (int i = 0; i < POOL_SIZE; i++) {
        pthread_create(&threads[i], NULL, worker_thread, NULL);
        pthread_detach(threads[i]);
    }
}

void pool_submit(void (*func)(void *), void *arg)
{
    pthread_mutex_lock(&pool.mutex);

    while (pool.count == QUEUE_SIZE) {
        pthread_cond_wait(&pool.not_full, &pool.mutex);
    }

    pool.queue[pool.rear].function = func;
    pool.queue[pool.rear].argument = arg;
    pool.rear = (pool.rear + 1) % QUEUE_SIZE;
    pool.count++;
    pthread_cond_signal(&pool.not_empty);

    pthread_mutex_unlock(&pool.mutex);
}

// 示例任务
void example_task(void *arg)
{
    int id = *(int *)arg;
    printf("Task %d executing in thread\n", id);
    sleep(1);
}

int main(void)
{
    pool_init();

    for (int i = 0; i < 10; i++) {
        int *id = malloc(sizeof(int));
        *id = i;
        pool_submit(example_task, id);
    }

    sleep(5);  // 等待任务完成
    pool.shutdown = 1;
    pthread_cond_broadcast(&pool.not_empty);

    return 0;
}
```

## 文件处理

大文件分块并行处理，提高 I/O 效率。

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

#define NUM_THREADS 4

typedef struct {
    FILE *file;
    long start;
    long end;
    int thread_id;
} FileChunk;

void *process_chunk(void *arg)
{
    FileChunk *chunk = (FileChunk *)arg;

    fseek(chunk->file, chunk->start, SEEK_SET);

    char buffer[1024];
    long bytes_read = 0;
    long total = chunk->end - chunk->start;

    while (bytes_read < total && fgets(buffer, sizeof(buffer), chunk->file)) {
        // 处理每一行数据
        printf("Thread %d: %s", chunk->thread_id, buffer);
        bytes_read += strlen(buffer);
    }

    free(chunk);
    return NULL;
}

void process_file_parallel(const char *filename)
{
    FILE *file = fopen(filename, "r");
    if (!file) return;

    fseek(file, 0, SEEK_END);
    long file_size = ftell(file);
    rewind(file);

    pthread_t threads[NUM_THREADS];
    long chunk_size = file_size / NUM_THREADS;

    for (int i = 0; i < NUM_THREADS; i++) {
        FileChunk *chunk = malloc(sizeof(FileChunk));
        chunk->file = file;
        chunk->start = i * chunk_size;
        chunk->end = (i == NUM_THREADS - 1) ? file_size : (i + 1) * chunk_size;
        chunk->thread_id = i;

        pthread_create(&threads[i], NULL, process_chunk, chunk);
    }

    for (int i = 0; i < NUM_THREADS; i++) {
        pthread_join(threads[i], NULL);
    }

    fclose(file);
}

int main(int argc, char *argv[])
{
    if (argc > 1) {
        process_file_parallel(argv[1]);
    }
    return 0;
}
```

## 数据库连接池

管理数据库连接，避免频繁创建连接的开销。

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <mysql/mysql.h>  // 示例使用 MySQL

#define POOL_SIZE 10

typedef struct {
    MYSQL *connections[POOL_SIZE];
    int available[POOL_SIZE];
    pthread_mutex_t mutex;
    pthread_cond_t cond;
} DBPool;

static DBPool db_pool;

void db_pool_init(const char *host, const char *user,
                  const char *pass, const char *db)
{
    pthread_mutex_init(&db_pool.mutex, NULL);
    pthread_cond_init(&db_pool.cond, NULL);

    for (int i = 0; i < POOL_SIZE; i++) {
        db_pool.connections[i] = mysql_init(NULL);
        mysql_real_connect(db_pool.connections[i],
                           host, user, pass, db, 3306, NULL, 0);
        db_pool.available[i] = 1;
    }
}

MYSQL *db_pool_get_connection(void)
{
    pthread_mutex_lock(&db_pool.mutex);

    int idx = -1;
    while (1) {
        for (int i = 0; i < POOL_SIZE; i++) {
            if (db_pool.available[i]) {
                idx = i;
                db_pool.available[i] = 0;
                break;
            }
        }
        if (idx >= 0) break;
        pthread_cond_wait(&db_pool.cond, &db_pool.mutex);
    }

    pthread_mutex_unlock(&db_pool.mutex);
    return db_pool.connections[idx];
}

void db_pool_release_connection(MYSQL *conn)
{
    pthread_mutex_lock(&db_pool.mutex);

    for (int i = 0; i < POOL_SIZE; i++) {
        if (db_pool.connections[i] == conn) {
            db_pool.available[i] = 1;
            pthread_cond_signal(&db_pool.cond);
            break;
        }
    }

    pthread_mutex_unlock(&db_pool.mutex);
}
```

## GUI 应用中的后台任务

保持界面响应，将耗时操作放到后台线程。

```c
#include <stdio.h>
#include <pthread.h>

// 模拟 GUI 框架的回调
typedef void (*ProgressCallback)(int percent);

typedef struct {
    const char *filename;
    ProgressCallback on_progress;
} DownloadTask;

void *download_file(void *arg)
{
    DownloadTask *task = (DownloadTask *)arg;

    // 模拟下载过程
    for (int i = 0; i <= 100; i += 10) {
        // 模拟网络延迟
        usleep(500000);

        // 更新进度（通过回调通知主线程/GUI）
        if (task->on_progress) {
            task->on_progress(i);
        }
    }

    printf("Download complete: %s\n", task->filename);
    free(task);
    return NULL;
}

void start_download(const char *filename, ProgressCallback callback)
{
    DownloadTask *task = malloc(sizeof(DownloadTask));
    task->filename = filename;
    task->on_progress = callback;

    pthread_t tid;
    pthread_create(&tid, NULL, download_file, task);
    pthread_detach(tid);
}

// 主线程中的进度更新
void on_progress_update(int percent)
{
    printf("Progress: %d%%\n", percent);
    // 实际应用中，这里会更新 GUI 进度条
}

int main(void)
{
    start_download("large_file.zip", on_progress_update);

    // 主线程继续处理 GUI 事件
    sleep(10);
    return 0;
}
```

## 定时任务调度

使用条件变量的超时功能实现定时任务。

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <time.h>

static pthread_mutex_t timer_mutex = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t timer_cond = PTHREAD_COND_INITIALIZER;
static int timer_running = 1;

void *timer_thread(void *arg)
{
    int interval_ms = *(int *)arg;
    struct timespec ts;

    while (timer_running) {
        clock_gettime(CLOCK_REALTIME, &ts);
        ts.tv_sec += interval_ms / 1000;
        ts.tv_nsec += (interval_ms % 1000) * 1000000;

        pthread_mutex_lock(&timer_mutex);
        int ret = pthread_cond_timedwait(&timer_cond, &timer_mutex, &ts);
        pthread_mutex_unlock(&timer_mutex);

        if (ret == ETIMEDOUT) {
            printf("Timer fired! Executing scheduled task...\n");
            // 执行定时任务
        }
    }

    return NULL;
}

void stop_timer(void)
{
    timer_running = 0;
    pthread_cond_broadcast(&timer_cond);
}

int main(void)
{
    pthread_t tid;
    int interval = 2000;  // 2秒

    pthread_create(&tid, NULL, timer_thread, &interval);

    sleep(10);
    stop_timer();
    pthread_join(tid, NULL);

    return 0;
}
```

## 缓存系统

使用读写锁实现高效的缓存读写。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <pthread.h>

#define CACHE_SIZE 1000

typedef struct {
    int key;
    char value[256];
    int valid;
} CacheEntry;

static CacheEntry cache[CACHE_SIZE];
static pthread_rwlock_t cache_lock = PTHREAD_RWLOCK_INITIALIZER;

int cache_get(int key, char *value, size_t size)
{
    pthread_rwlock_rdlock(&cache_lock);

    for (int i = 0; i < CACHE_SIZE; i++) {
        if (cache[i].valid && cache[i].key == key) {
            strncpy(value, cache[i].value, size - 1);
            value[size - 1] = '\0';
            pthread_rwlock_unlock(&cache_lock);
            return 1;  // 命中
        }
    }

    pthread_rwlock_unlock(&cache_lock);
    return 0;  // 未命中
}

void cache_put(int key, const char *value)
{
    pthread_rwlock_wrlock(&cache_lock);

    // 查找空位或更新现有项
    int idx = -1;
    for (int i = 0; i < CACHE_SIZE; i++) {
        if (!cache[i].valid) {
            idx = i;
            break;
        }
        if (cache[i].key == key) {
            idx = i;
            break;
        }
    }

    if (idx >= 0) {
        cache[idx].key = key;
        strncpy(cache[idx].value, value, sizeof(cache[idx].value) - 1);
        cache[idx].value[sizeof(cache[idx].value) - 1] = '\0';
        cache[idx].valid = 1;
    }

    pthread_rwlock_unlock(&cache_lock);
}

int main(void)
{
    // 初始化缓存
    memset(cache, 0, sizeof(cache));

    // 写入数据
    cache_put(1, "User: Alice");
    cache_put(2, "User: Bob");

    // 读取数据
    char value[256];
    if (cache_get(1, value, sizeof(value))) {
        printf("Cache hit: %s\n", value);
    }

    pthread_rwlock_destroy(&cache_lock);
    return 0;
}
```
